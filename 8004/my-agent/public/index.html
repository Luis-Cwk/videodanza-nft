Prueba rapida

curl -s https://my-agent-tau.vercel.app/.well-known/agent-card.json | head -c 300

# Salud, gratis
curl -s https://my-agent-tau.vercel.app/health

# Tools MCP disponibles
curl -s -X POST https://my-agent-tau.vercel.app/mcp \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"tools/list","params":{}}'

# Chat A2A
curl -s -X POST https://my-agent-tau.vercel.app/a2a \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"message/send","params":{"message":{"role":"user","parts":[{"type":"text","text":"Hola, quien eres?"}]}}}'
